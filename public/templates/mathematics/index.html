Quick Start: Symbolic Math Workflow

From problem statement to elegant solution in 5 steps.

1

Define Symbols & Expressions

Create symbolic variables, define mathematical functions, set up expressions for analysis using SymPy's symbolic engine.

# Initialize symbolic computation from sympy import * from sympy.parsing.sympy_parser import parse_expr # Define symbolic variables x, y, z = symbols('x y z', real=True) n = symbols('n', integer=True, positive=True) # Define mathematical expressions f = sin(x)**2 + cos(x)**2 # Trigonometric identity g = x**3 * exp(-x) # Function for calculus h = log(x) + sqrt(y) - z**2 # Multivariable expression print("Expressions defined:") print(f"f(x) = {f}") print(f"g(x) = {g}") print(f"h(x,y,z) = {h}")
2

Simplify & Transform Expressions

Apply algebraic identities, factor polynomials, expand expressions, rationalize denominators, and convert to different forms.

# Expression simplification and transformation # Simplify trigonometric identity f_simplified = simplify(f) print(f"Simplify sin²(x) + cos²(x):") print(f" Result: {f_simplified}") # Should be 1 # Factor polynomial poly = x**3 - 6*x**2 + 11*x - 6 factored = factor(poly) print(f"\nFactor x³ - 6x² + 11x - 6:") print(f" Result: {factored}") # Expand and collect terms expanded = expand((x + y)**5) collected = expanded.collect(x) print(f"\nExpand (x+y)⁵:") print(f" Terms in x: {collected}") # Rational simplification expr = (x**2 - 1)/(x - 1) rational_simplified = simplify(expr) print(f"\nSimplify (x²-1)/(x-1):") print(f" Result: {rational_simplified}")
3

Differentiation & Integration

Compute partial derivatives, gradients, definite/indefinite integrals, find critical points, and solve differential equations analytically.

# Calculus operations # First and higher-order derivatives g_prime = diff(g, x) # First derivative: d/dx [x³e^(-x)] g_double_prime = diff(g, x, 2) # Second derivative print("Derivatives of g(x) = x³e^(-x):") print(f" g'(x) = {g_prime}") print(f" g''(x) = {g_double_prime}") # Partial derivatives of multivariable function dh_dx = diff(h, x) dh_dy = diff(h, y) print(f"\nPartial derivatives of h(x,y,z) = ln(x) + √y - z²:") print(f" ∂h/∂x = {dh_dx}") print(f" ∂h/∂y = {dh_dy}") # Indefinite integral integral_g = integrate(g, x) print(f"\nIndefinite integral of g(x):") print(f" ∫g(x)dx = {integral_g} + C") # Definite integral definite_integral = integrate(g, (x, 0, oo)) print(f"\nDefinite integral from 0 to ∞:") print(f" ∫₀^∞ g(x)dx = {definite_integral}") # Find critical points critical_points = solve(g_prime, x) print(f"\nCritical points of g(x):") for point in critical_points: print(f" x = {point}, g(x) = {g.subs(x, point).evalf()}")
4

Solve Equations & Systems

Find roots of polynomials, solve systems of linear/nonlinear equations, differential equations analytically or numerically.

# Equation solving # Polynomial roots quadratic = Eq(x**2 + 5*x + 6, 0) roots_quadratic = solve(quadratic, x) print("Solve x² + 5x + 6 = 0:") print(f" Solutions: {roots_quadratic}") # System of nonlinear equations eq1 = Eq(x**2 + y**2, 1) # Circle eq2 = Eq(y, x**2) # Parabola system_solution = solve((eq1, eq2), (x, y)) print(f"\nSolve system (circle & parabola):") for sol in system_solution: print(f" x = {sol[0]}, y = {sol[1]}") # Differential equation: y'' + y = 0 y = Function('y') ode = Eq(diff(y(x), x, x) + y(x), 0) ode_solution = dsolve(ode, y(x)) print(f"\nSolve ODE: y'' + y = 0") print(f" General solution: {ode_solution}") # Initial value problem ivp_solution = dsolve(ode, y(x), ics={y(0): 1, diff(y(x), x).subs(x, 0): 0}) print(f" IVP solution (y(0)=1, y'(0)=0): {ivp_solution}") # Matrix operations A = Matrix([[1, 2], [3, 4]]) eigenvalues = A.eigenvals() print(f"\nMatrix A eigenvalues: {list(eigenvalues.keys())}")
5

Visualize Results

Generate publication-quality plots of functions, phase portraits, contour maps, 3D surfaces, complex plane visualizations, and export to LaTeX/PDF.

# Mathematical visualization import matplotlib.pyplot as plt import numpy as np from mpl_toolkits.mplot3d import Axes3D # Create figure with multiple subplots fig = plt.figure(figsize=(14, 10)) # Plot 1: Function curve ax1 = fig.add_subplot(221) x_vals = np.linspace(0.01, 5, 400) g_func = lambda x: x**3 * np.exp(-x) ax1.plot(x_vals, [g_func(v) for v in x_vals], 'b-', linewidth=2, label='g(x)') ax1.fill_between(x_vals, [g_func(v) for v in x_vals], alpha=0.2) ax1.set_title('Function: $g(x) = x^3 e^{-x}$', fontsize=12) ax1.grid(True, alpha=0.3) ax1.legend() # Plot 2: Contour plot of h(x,y) ax2 = fig.add_subplot(222) X, Y = np.meshgrid(np.linspace(0.1, 3, 100), np.linspace(0.1, 3, 100)) Z = np.log(X) + np.sqrt(Y) - 2**2 # Fix z=2 for visualization contour = ax2.contourf(X, Y, Z, levels=20, cmap='viridis') plt.colorbar(contour, ax=ax2, label='h(x,y,2)') ax2.set_title('Contour: $h(x,y) = \ln(x) + \sqrt{y} - z^2$', fontsize=12) # Plot 3: 3D surface ax3 = fig.add_subplot(223, projection='3d') X3, Y3 = np.meshgrid(np.linspace(-2, 2, 50), np.linspace(-2, 2, 50)) Z3 = X3**2 + Y3**2 # Paraboloid ax3.plot_surface(X3, Y3, Z3, cmap='plasma', alpha=0.8) ax3.set_title('Surface: $z = x^2 + y^2$', fontsize=12) # Plot 4: Complex function ax4 = fig.add_subplot(224) theta = np.linspace(0, 2*np.pi, 400) complex_func = np.exp(1j * theta) # Unit circle ax4.plot(complex_func.real, complex_func.imag, 'r-', linewidth=2) ax4.scatter([1], [0], c='green', s=100, zorder=5, label='e^(i·0)=1') ax4.set_aspect('equal') ax4.grid(True, alpha=0.3) ax4.set_title("Euler's Formula: $e^{i\\theta}$", fontsize=12) ax4.legend() plt.tight_layout() plt.savefig('math_visualization.png', dpi=300, bbox_inches='tight', facecolor='#0a0a0f') plt.show() # Export symbolic result to LaTeX latex_output = latex(integral_g) print(f"\nLaTeX output: {latex_output}")
