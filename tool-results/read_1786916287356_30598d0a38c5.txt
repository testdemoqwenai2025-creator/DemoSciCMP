   909→// ============================================================================
   910→// URL ROUTING CONFIGURATION
   911→// ============================================================================
   912→
   913→const TEMPLATE_SLUGS: Record<string, string> = {
   914→  'blast': 'blast-sequence-analysis',
   915→  'docking': 'molecular-docking-workflow',
   916→  'training': 'transformer-training-pipeline',
   917→  'stats': 'statistical-analysis-suite',
   918→  'statistics': 'statistical-analysis-suite',
   919→  'viz': 'visualization-templates',
   920→  'visualization': 'visualization-templates',
   921→  'guide': 'create-template-guide',
   922→  'template-guide': 'create-template-guide',
   923→};
   924→
   925→function parseHashRoute(hash: string): { templateId: string | null; category: TemplateCategory | 'all'; section?: string } {
   926→  const cleanHash = hash.replace(/^#\/?/, '');
   927→  
   928→  // Check for special sections
   929→  if (cleanHash === 'capabilities' || cleanHash === 'core-capabilities') {
   930→    return { templateId: null, category: 'all', section: 'capabilities' };
   931→  }
   932→  if (cleanHash === 'use-cases') {
   933→    return { templateId: null, category: 'all', section: 'use-cases' };
   934→  }
   935→  if (cleanHash === 'free-tier' || cleanHash === 'resources') {
   936→    return { templateId: null, category: 'all', section: 'free-tier' };
   937→  }
   938→  
   939→  const templatesMatch = cleanHash.match(/templates\/(?:([^/]+)\/)?([^/]+)\/?$/);
   940→  
   941→  if (templatesMatch) {
   942→    const templateSlug = templatesMatch[2];
   943→    
   944→    // Direct ID match
   945→    const directMatch = templates.find(t => t.id === templateSlug);
   946→    if (directMatch) return { templateId: directMatch.id, category: directMatch.category };
   947→    
   948→    // Slug mapping
   949→    const mappedId = TEMPLATE_SLUGS[templateSlug.toLowerCase()];
   950→    if (mappedId) {
   951→      const template = templates.find(t => t.id === mappedId);
   952→      if (template) return { templateId: template.id, category: template.category };
   953→    }
   954→  }
   955→  
   956→  return { templateId: null, category: 'all' };
   957→}
