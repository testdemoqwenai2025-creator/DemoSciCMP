  1412→        function performSearch() {
  1413→            const query = document.getElementById('searchInput').value.toLowerCase().trim();
  1414→            const cards = document.querySelectorAll('.paper-card');
  1415→            
  1416→            if (!query && currentFilter === 'all') {

  1418→                return;
  1419→            }
  1420→
  1421→            cards.forEach(card => {
  1422→                const title = card.querySelector('.paper-title').textContent.toLowerCase();
  1423→                const authors = card.querySelector('.paper-authors').textContent.toLowerCase();
  1424→                const abstract = card.querySelector('.paper-abstract').textContent.toLowerCase();
  1425→                const topics = card.dataset.topics || '';
  1426→                
  1427→                const matchesText = title.includes(query) || authors.includes(query) || abstract.includes(query);
  1428→                const matchesFilter = currentFilter === 'all' || topics.includes(currentFilter);
  1429→                
  1430→                if ((!query || matchesText) && matchesFilter) {
  1431→                    card.style.display = '';
  1432→                } else {
  1433→                    card.style.display = 'none';
  1434→                }
  1435→            });
  1436→
  1437→            showNotification(`🔍 Searching: "${query || 'All'}" | Filter: ${currentFilter}`);
  1438→        }
